def read_html_parkings():
    """Читает парковки из HTML файла"""
    html_files = [f for f in os.listdir('.') if f.endswith('.html')]

    if not html_files:
        print("HTML файлы не найдены.")
        return []

    html_file = html_files[0]
    print(f"Читаю файл: {html_file}")

    try:
        with open(html_file, 'r', encoding='utf-8') as f:
            content = f.read()

        _markersMeta
        pattern = r'_markersMeta\s*=\s*\[(.*?)\];'
        match = re.search(pattern, content, re.DOTALL)

        if not match:
            print("Не найден массив _markersMeta")
            return []

        markers_text = match.group(1)

        # Простой парсинг объектов
        objects = re.findall(r'\{[^}]+\}', markers_text)

        parkings = []

        for i, obj in enumerate(objects):
            try:
                # Извлекаем поля
                lat_match = re.search(r'lat\s*:\s*([\d.]+)', obj)
                lon_match = re.search(r'lon\s*:\s*([\d.]+)', obj)
                key_match = re.search(r'spotKey\s*:\s*[\'"]([^\'"]+)[\'"]', obj)
                title_match = re.search(r'title\s*:\s*[\'"]([^\'"]+)[\'"]', obj)

                if lat_match and lon_match:
                    lat = float(lat_match.group(1))
                    lon = float(lon_match.group(1))
                    key = key_match.group(1) if key_match else f"parking_{i + 1}"
                    title = title_match.group(1) if title_match else f"Парковка {i + 1}"

                    parkings.append({
                        'key': key,
                        'title': title,
                        'lat': lat,
                        'lon': lon
                    })
            except:
                continue

        return parkings

    except Exception as e:
        print(f"Ошибка чтения HTML: {e}")
        return []


def select_parking(parkings):
    """Выбор парковки"""
    if not parkings:
        print("Нет парковок для выбора")
        return None

    print("\n" + "=" * 50)
    print("ВЫБОР ПАРКОВКИ")
    print("=" * 50)

    for i, parking in enumerate(parkings, 1):
        print(f"{i}. {parking['title']}")
        print(f"   Ключ: {parking['key']}")
        print()

    try:
        choice = int(input(f"Выберите парковку (1-{len(parkings)}): "))
        if 1 <= choice <= len(parkings):
            return parkings[choice - 1]
        else:
            print("Неверный выбор")
            return None
    except ValueError:
        print("Введите число")
        return None


def main():
    """Главная функция"""
    print("\nЗагрузка данных о парковках...")
    parkings = read_html_parkings()

    if not parkings:
        print("Не удалось загрузить парковки")
        print("Убедитесь, что в папке есть HTML файл с картой парковок")
        return

    # Выбор парковки
    selected_parking = select_parking(parkings)
    if not selected_parking:
        return


if __name__ == "__main__":
    main()
